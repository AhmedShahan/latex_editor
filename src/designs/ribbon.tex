\definecolor{ribbon}{RGB}{200,60,80}
\definecolor{ribbondark}{RGB}{140,30,50}
\definecolor{titlecol}{RGB}{50,50,60}
\definecolor{numcol}{RGB}{255,255,255}

% ---------- Section: বড় ফিতা ----------
\newcommand{\sectionfmt}[1]{%
  \noindent
  \begin{tikzpicture}[baseline=(base.base)]
    \coordinate (base) at (0,0);
    \fill[ribbon] (-0.35cm,0.6cm) -- (0.45cm,0.6cm) -- (0.45cm,-0.55cm)
                  -- (0.05cm,-0.25cm) -- (-0.35cm,-0.55cm) -- cycle;
    \fill[ribbondark] (0.45cm,0.6cm) -- (0.45cm,0.45cm) -- (0.6cm,0.6cm) -- cycle;
    \node[inner sep=0pt] at (0.05cm,0.1cm) {\sffamily\bfseries\Large\color{numcol}\thesection};
    \node[anchor=west, inner sep=0pt] at (0.8cm,0.05cm)
      {\sffamily\bfseries\LARGE\color{titlecol}#1};
    \draw[ribbon!60, dotted, line width=1pt] (0.8cm,-0.4cm) -- (\linewidth-0.35cm,-0.4cm);
  \end{tikzpicture}\par}

% ---------- Subsection: মাঝারি ফিতা ----------
\newcommand{\subsectionfmt}[1]{%
  \noindent
  \begin{tikzpicture}[baseline=(base.base)]
    \coordinate (base) at (0,0);
    \fill[ribbon] (-0.3cm,0.45cm) -- (0.55cm,0.45cm) -- (0.55cm,-0.4cm)
                  -- (0.12cm,-0.18cm) -- (-0.3cm,-0.4cm) -- cycle;
    \fill[ribbondark] (0.55cm,0.45cm) -- (0.55cm,0.34cm) -- (0.66cm,0.45cm) -- cycle;
    \node[inner sep=0pt] at (0.12cm,0.06cm) {\sffamily\bfseries\normalsize\color{numcol}\thesubsection};
    \node[anchor=west, inner sep=0pt] at (0.85cm,0.03cm)
      {\sffamily\bfseries\Large\color{titlecol}#1};
    \draw[ribbon!60, dotted, line width=0.8pt] (0.85cm,-0.3cm) -- (\linewidth-0.3cm,-0.3cm);
  \end{tikzpicture}\par}

% ---------- Subsubsection: ছোট ফিতা ----------
\newcommand{\subsubsectionfmt}[1]{%
  \noindent
  \begin{tikzpicture}[baseline=(base.base)]
    \coordinate (base) at (0,0);
    \fill[ribbon] (-0.25cm,0.33cm) -- (0.7cm,0.33cm) -- (0.7cm,-0.3cm)
                  -- (0.22cm,-0.12cm) -- (-0.25cm,-0.3cm) -- cycle;
    \fill[ribbondark] (0.7cm,0.33cm) -- (0.7cm,0.25cm) -- (0.78cm,0.33cm) -- cycle;
    \node[inner sep=0pt] at (0.22cm,0.04cm) {\sffamily\bfseries\footnotesize\color{numcol}\thesubsubsection};
    \node[anchor=west, inner sep=0pt] at (0.95cm,0.02cm)
      {\sffamily\bfseries\large\color{titlecol}#1};
    \draw[ribbon!60, dotted, line width=0.6pt] (0.95cm,-0.22cm) -- (\linewidth-0.25cm,-0.22cm);
  \end{tikzpicture}\par}

\titleformat{\section}[block]{}{}{0pt}{\sectionfmt}
\titleformat{\subsection}[block]{}{}{0pt}{\subsectionfmt}
\titleformat{\subsubsection}[block]{}{}{0pt}{\subsubsectionfmt}

\titlespacing*{\subsubsection}{0.25cm}{2ex plus 1ex minus .2ex}{2ex plus .2ex}
